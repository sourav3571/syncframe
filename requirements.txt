# SyncFrame project requirements (platform-agnostic)
# These are environment and CLI dependencies (not npm package lock file).

# 1. System / toolchain
Node.js >= 18.0.0 (recommended v20 LTS)
Rust stable >= 1.75

# 2. CLI packages
npm install -g @tauri-apps/cli

# 3. Native tools
ffmpeg (ffmpeg + ffprobe in PATH)

# 4. Platform-specific prerequisites
# Windows: Visual Studio 2022 Build Tools (C++), WebView2 runtime
# macOS: Xcode Command Line Tools
# Linux: libwebkit2gtk-4.1, libssl-dev, libgtk-3-dev (Tauri prerequisites)

# 5. Run commands
npm install
npm run tauri dev
npm run tauri build
